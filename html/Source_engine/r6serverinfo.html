event RestartServer()
{
    if (m_GameInfo!=none)
    {
        m_GameInfo.AbortScoreSubmission();
        m_GameInfo.bChangeLevels = true;
        m_GameInfo.m_bChangedServerConfig = true;
        m_GameInfo.SetJumpingMaps(true, 0);
        m_GameInfo.RestartGameMgr();
    }
}
